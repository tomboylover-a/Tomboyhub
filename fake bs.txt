local function HookBeatTracker()
    local modules = ReplicatedStorage:FindFirstChild("Modules")
    if not modules then return end
    local gameplay = modules:FindFirstChild("Gameplay")
    if not gameplay then return end
    local beatTrackerModule = gameplay:FindFirstChild("BeatTracker")
    if not beatTrackerModule then return end
    
    local success, BeatTracker = pcall(require, beatTrackerModule)
    if not success or not BeatTracker then return end
    
    if _G.BeatTrackerHooked then return end
    _G.BeatTrackerHooked = true

    local oldUpdate = BeatTracker.update
    BeatTracker.update = function(self, dt)
        local v1 = tick()
        local beatHappened = false
        if self.nextBeat < v1 then
            beatHappened = true
        end

        oldUpdate(self, dt)

        if beatHappened and _G.MusicianAutoPlay then
            task.defer(function()
                if _G.MusicianAutoPlay and self.songSource and self.songSource.IsPlaying then
                    self:hit()
                end
            end)
        end
    end
end

pcall(HookBeatTracker)

function L.updateAttackSpeed()
    if not L.attackSpeedEnabled then
        if L.attackSpeedConn then
            L.attackSpeedConn:Disconnect()
            L.attackSpeedConn = nil
        end
        local char = LocalPlayer.Character
        if char then
            for _, tool in char:GetChildren() do
                if tool:IsA("Tool") then
                    local buff = tool:FindFirstChild("SwingSpeedBuff")
                    if buff then buff:Destroy() end
                end
            end
        end
        return
    end

    if not L.attackSpeedConn then
        L.attackSpeedConn = RunService.Heartbeat:Connect(function()
            if not L.attackSpeedEnabled then return end
            local char = LocalPlayer.Character
            if not char then return end
            for _, tool in char:GetChildren() do
                if tool:IsA("Tool") and tool:GetAttribute("Melee") then
                    local buff = tool:FindFirstChild("SwingSpeedBuff")
                    if not buff then
                        buff = Instance.new("NumberValue")
                        buff.Name = "SwingSpeedBuff"
                        buff.Parent = tool
                    end
                    buff.Value = L.attackSpeedMultiplier
                end
            end
        end)
    end
end

function L.toggleAttackSpeed(state)
    L.attackSpeedEnabled = state
    L.updateAttackSpeed()
end

local function isZombieDead(zombieModel)
    if not zombieModel or not zombieModel.Parent then return true end
    local humanoid = zombieModel:FindFirstChildOfClass("Humanoid")
    local orig = zombieModel:FindFirstChild("Orig")
    if not humanoid and orig and orig.Value then
        humanoid = orig.Value:FindFirstChildOfClass("Humanoid")
    end
    if humanoid and humanoid.Health <= 0 then
        return true
    end
    return false
end

local function getZombieClassification(child)
    local name = child.Name:lower()
    local isBomber = child:FindFirstChild("Barrel") ~= nil or name:find("bomber") or name:find("barrel")
    local isIgniter = child:FindFirstChild("Lantern") ~= nil or child:FindFirstChild("Torch") ~= nil or name:find("igniter") or name:find("lantern")
    local isCuirassier = child:FindFirstChild("Sword") ~= nil or name:find("cuirassier") or name:find("sword")
    local isZapper = child:FindFirstChild("Axe") ~= nil or name:find("zapper") or name:find("axe")
    
    local zombieRef = nil
    local origin = child:FindFirstChild("Orig")
    if origin and origin.Value then
        zombieRef = origin.Value
    end

    local ws = 16
    if zombieRef and zombieRef:FindFirstChild("Zombie") then
        ws = zombieRef.Zombie.WalkSpeed or 16
    elseif child:FindFirstChild("Humanoid") then
        ws = child.Humanoid.WalkSpeed or 16
    end

    local isRunner = (ws >= 20) or name:find("runner") or name:find("fast")
    local isShambler = not (isBomber or isRunner or isIgniter or isCuirassier or isZapper)

    return {
        shambler = isShambler,
        runner = isRunner,
        bomber = isBomber,
        cuirassier = isCuirassier,
        zapper = isZapper,
        igniter = isIgniter
    }
end

local function getZombieModel(part)
    if not part then return nil end
    local model = part.Parent
    for i = 1, 5 do
        if not model then break end
        if model:IsA("Model") and (model.Name == "m_Zombie" or model:FindFirstChild("Orig")) then
            return model
        end
        model = model.Parent
    end
    return nil
end

local function getZombieHead(zombieModel)
    if not zombieModel then return nil end
    for _, part in zombieModel:GetChildren() do
        if part.Name == "Head" and (part:IsA("Part") or part:IsA("MeshPart")) then
            return part
        end
    end
    return nil
end

local function getRemoteEvent(self)
    if self and self.remoteEvent then
        return self.remoteEvent
    end
    local char = LocalPlayer.Character
    if char then
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") then
                local remote = child:FindFirstChild("RemoteEvent") or child:FindFirstChild("Remote")
                if remote then return remote end
            end
        end
    end
    return nil
end

function L.unifiedBayonetHitCheck(self, origin, direction, raycastParams, hitEntities)
    local rayResult = workspace:Raycast(origin, direction, raycastParams)
    if rayResult and L.tomboyHitterEnabled then
        local hitPart = rayResult.Instance
        local zombieModel = getZombieModel(hitPart)
        if zombieModel then
            if isZombieDead(zombieModel) then return 0 end
            local now = tick()
            if L.lastZombieHitTime[zombieModel] and now - L.lastZombieHitTime[zombieModel] < L.ZOMBIE_HIT_COOLDOWN then
                return 0
            end
            local orig = zombieModel:FindFirstChild("Orig")
            if orig then
                local head = getZombieHead(zombieModel) or hitPart
                local zombieRef = orig.Value
                
                local targetPart = L.headshotEnabled and head or hitPart
                local partName = (targetPart == head) and "Head" or "Torso"
                local realHitPos = targetPart.Position
                local sendHitPos = realHitPos

                if L.removeBloodEnabled then
                    local rootPart = zombieModel:FindFirstChild("HumanoidRootPart") or zombieModel:FindFirstChild("Torso")
                    if rootPart then
                        sendHitPos = rootPart.Position + Vector3.new(0, -9999, 0)
                    end
                end

                local char = LocalPlayer.Character
                local headPart = char and char:FindFirstChild("Head")
                local Direction = headPart and (realHitPos - headPart.Position).Unit or Vector3.new(0, 1, 0)

                local remote = getRemoteEvent(self)
                if remote then
                    pcall(function()
                        remote:FireServer("Bayonet_HitZombie", zombieRef, realHitPos, true, partName)
                        remote:FireServer("HitZombieM", zombieRef, sendHitPos, true, head, partName, Direction)
                    end)
                end

                L.lastZombieHitTime[zombieModel] = now
                return 1
            end
        end
    end
    
    if L.originalBayonetHitCheck then
        return L.originalBayonetHitCheck(self, origin, direction, raycastParams, hitEntities)
    end
    return 0
end

function L.updateHitHooks()
    local rs = ReplicatedStorage
    local weapons = rs:FindFirstChild("Modules") and rs.Modules:FindFirstChild("Weapons")
    local canHookFunction = type(hookfunction) == "function"

    if weapons and not L.hookInstalled then
        local success, FlintLock = pcall(require, weapons:FindFirstChild("Flintlock"))
        if success and FlintLock and FlintLock.BayonetHitCheck then
            if canHookFunction then
                L.originalBayonetHitCheck = hookfunction(FlintLock.BayonetHitCheck, L.unifiedBayonetHitCheck)
            else
                L.originalBayonetHitCheck = FlintLock.BayonetHitCheck
                FlintLock.BayonetHitCheck = L.unifiedBayonetHitCheck
            end
        end

        local success2, MeleeBase = pcall(require, weapons:FindFirstChild("MeleeBase"))
        if success2 and MeleeBase and MeleeBase.MeleeHitCheck then
            if canHookFunction then
                L.originalMeleeHitCheck = hookfunction(MeleeBase.MeleeHitCheck, L.unifiedMeleeHitCheck)
            else
                L.originalMeleeHitCheck = MeleeBase.MeleeHitCheck
                MeleeBase.MeleeHitCheck = L.unifiedMeleeHitCheck
            end
        end

        L.hookInstalled = true
    end
end

local function isValidZombieType(child)
    if isZombieDead(child) then return false end
    local types = getZombieClassification(child)
    if types.shambler and L.targetShambler then return true end
    if types.runner and L.targetRunner then return true end
    if types.bomber and L.targetBomber then return true end
    if types.cuirassier and L.targetCuirassier then return true end
    if types.zapper and L.targetZapper then return true end
    if types.igniter and L.targetIgniter then return true end
    return false
end

RunService.Heartbeat:Connect(function()
    pcall(function()
        local function processFolder(folder)
            if not folder then return end
            for _, child in pairs(folder:GetChildren()) do
                if child:IsA("Model") and (child.Name == "m_Zombie" or child:FindFirstChild("Orig") or child:FindFirstChild("Humanoid")) then
                    local hrp = child:FindFirstChild("HumanoidRootPart") or child:FindFirstChild("Torso")
                    if hrp and hrp:IsA("BasePart") then
                        if L.hitboxExpanderEnabled and not isZombieDead(child) then
                            local targetSize = v3new(L.hitboxSize, L.hitboxSize, L.hitboxSize)
                            local targetTransparency = L.hitboxVisualsEnabled and ((L.hitboxTransparency or 70) / 100) or 1
                            local targetColor = L.hitboxColor or Color3.fromRGB(255, 0, 0)
                            
                            if hrp.Size ~= targetSize or hrp.Transparency ~= targetTransparency or hrp.Color ~= targetColor then
                                hrp.Size = targetSize
                                hrp.Transparency = targetTransparency
                                hrp.Color = targetColor
                                hrp.CanCollide = false
                                hrp.Massless = true
                            end
                        else
                            if hrp.Name == "HumanoidRootPart" then
                                if hrp.Size ~= v3new(2, 2, 1) or hrp.Transparency ~= 1 then
                                    hrp.Size = v3new(2, 2, 1)
                                    hrp.Transparency = 1
                                end
                            elseif hrp.Name == "Torso" then
                                if hrp.Size ~= v3new(2, 2, 1) or hrp.Transparency ~= 0 then
                                    hrp.Size = v3new(2, 2, 1)
                                    hrp.Transparency = 0
                                end
                            end
                        end
                    end
                end
            end
        end
        processFolder(Workspace.Camera)
        processFolder(Workspace:FindFirstChild("Zombies"))
    end)
end)

local function getClosestFilteredTarget()
    local closestTarget = nil
    local shortestDistance = math.huge
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    pcall(function()
        local function scanFolder(folder)
            if not folder then return end
            for _, child in pairs(folder:GetChildren()) do
                if child:IsA("Model") and (child.Name == "m_Zombie" or child:FindFirstChild("Orig") or child:FindFirstChild("Humanoid")) then
                    local targetPartName = (L.aimbotTargetPart == "Body") and "HumanoidRootPart" or "Head"
                    local targetPart = child:FindFirstChild(targetPartName) or child:FindFirstChild("Head")
                    
                    if targetPart and isValidZombieType(child) then
                        local worldDist = (targetPart.Position - Camera.CFrame.Position).Magnitude
                        if worldDist <= L.aimbotMaxRange then
                            local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                            if onScreen then
                                local screenVector = Vector2.new(screenPos.X, screenPos.Y)
                                local distance = (screenVector - screenCenter).Magnitude
                                if distance < shortestDistance and distance <= L.aimbotFOV then
                                    shortestDistance = distance
                                    closestTarget = targetPart
                                end
                            end
                        end
                    end
                end
            end
        end
        scanFolder(Workspace.Camera)
        scanFolder(Workspace:FindFirstChild("Zombies"))
    end)
    
    return closestTarget
end

RunService.RenderStepped:Connect(function()
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    fovDrawingCircle.Position = screenCenter
    fovDrawingCircle.Radius = L.aimbotFOV
    fovDrawingCircle.Visible = L.aimbotFOVVisible or L.aimbotEnabled
    fovDrawingCircle.Filled = L.fovCircleFilled

    if not L.aimbotEnabled then
        lockBox.Visible = false
        return
    end
    
    local targetPart = getClosestFilteredTarget()
    if targetPart then
        Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetPart.Position)
        local model = targetPart.Parent
        local _, size = model:GetBoundingBox()
        local headPos = targetPart.Position
        local topScreen, topOnScreen = Camera:WorldToViewportPoint(headPos + Vector3.new(0, size.Y / 2, 0))
        local bottomScreen, bottomOnScreen = Camera:WorldToViewportPoint(headPos - Vector3.new(0, size.Y / 2, 0))
        
        if topOnScreen and bottomOnScreen then
            local height = math.abs(bottomScreen.Y - topScreen.Y)
            local width = height * 0.6
            lockBox.Size = Vector2.new(width, height)
            lockBox.Position = Vector2.new(topScreen.X - width / 2, topScreen.Y)
            lockBox.Visible = true
        else
            lockBox.Visible = false
        end
    else
        lockBox.Visible = false
    end
end)
local function ensureHighlight(target, fillColor, outlineColor)
    if not target or not target.Parent or isZombieDead(target) then return end
    local h = target:FindFirstChild("ESP_Highlight")
    if not h then
        h = Instance.new("Highlight")
        h.Name = "ESP_Highlight"
        h.Parent = target
        h.OutlineColor = outlineColor or Color3.fromRGB(255,255,255)
    end
    h.FillColor = fillColor or Color3.fromRGB(255,255,255)
end

local function removeHighlight(target)
    if not target then return end
    local h = target:FindFirstChild("ESP_Highlight")
    if h then h:Destroy() end
end

RunService.Heartbeat:Connect(function()
    pcall(function()
        local function processFolder(folder)
            if not folder then return end
            for _, child in pairs(folder:GetChildren()) do
                if child:IsA("Model") and (child.Name == "m_Zombie" or child:FindFirstChild("Orig") or child:FindFirstChild("Humanoid")) then

                        elseif espBToggled and types.bomber then
                            ensureHighlight(child, Color3.fromRGB(65, 105, 225))
                            highlighted = true
                        elseif espCuToggled and types.cuirassier then
                            ensureHighlight(child, Color3.fromRGB(65, 105, 225))
                            highlighted = true
                        elseif espZToggled and types.zapper then
                            ensureHighlight(child, Color3.fromRGB(255, 0, 255))
                            highlighted = true
                        elseif espIgniterToggled and types.igniter then
                            ensureHighlight(child, Color3.fromRGB(255, 69, 0))
                            highlighted = true
                        end

                            removeHighlight(child)
                        end

            end

        processFolder(Workspace:FindFirstChild("Zombies"))
    end)
end)
